(defn cypher-write-query?
  "True only for the one public Cypher write prefix the bounded parser accepts.
  Classification happens before the read cache/head path: CREATE must traverse
  the same verified-tenant, nonce, owner-record and CAS orchestration as
  datomic.transact, never the anonymous read route."
  [method body]
  (boolean (and (= method "cypher")
                (re-find #"(?i)^\s*CREATE\b" (str (:query body))))))

(defn- cypher-write-after-nonce-check [env body iss nonce db-name seen?]
  (cond
    seen?
    (js/Promise.resolve {:ok false :error "Unauthorized"
                         :message "CACAO nonce already used" :status 401})

    (not (valid-db-name? db-name))
    (js/Promise.resolve {:ok false :error "Bad Request"
                         :message "db_name required for Cypher CREATE" :status 400})

    :else
    (let [g (write-graph-name iss db-name)]
      (-> (owner/bind-owner! env g iss)
          (.then (fn [denial]
                   (if denial
                     (owner-write-denial denial)
                     (run-write env g "cypher" (assoc body :graph g) iss))))
          (.then (fn [resp] (transact-finish env iss nonce resp)))))))
