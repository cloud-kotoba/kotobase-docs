(defn handle-pull [request env]
  (-> (parse-json request)
      (.then (fn [body]
               (let [b (js-body->clj body)]
                 (if-not (string? (:entity b))
                   (bad-request "entity required")
                   (run-read env "pull" b (caller-did request))))))
      (.catch (fn [e] (handler-failure e)))))
